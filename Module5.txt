    Do
    i01 = i01 + 1
    If i01 > 20 Then Exit Sub
    Loop While Len(Cells(i01 * 7 + 5 - 7, 7).Value) < 1
    Y = i01
    For ik = 1 To i01 - i
    Call MOVE_line("UP", (Y))
    'Application.ScreenUpdating = False

    Next ik
    End If
Next i

End Sub

Sub PPU0(nom As Single)
i = nom
Dim I_ppu(15) As String
Dim D_ppu(15, 3) As Single

For i = 1 To 15
I_ppu(i) = ""
For j = 1 To 3
D_ppu(i, j) = 0
Next j
Next i

ch = 0
For i = 220 To 233
ai = Worksheets("CODE").Cells(i, 7).Value
a1 = Worksheets("CODE").Cells(i, 8).Value
a2 = Worksheets("CODE").Cells(i, 9).Value
a3 = Worksheets("CODE").Cells(i, 10).Value

If Len(ai) * Len(a1) * Len(a2) * Len(a3) > 0 Then
ch = ch + 1
I_ppu(ch) = ai
D_ppu(ch, 1) = a1
D_ppu(ch, 2) = a2
D_ppu(ch, 3) = a3
End If
Next i

For i = 1 To ch
UserForm12.ListBox1.AddItem I_ppu(i)
Next i
UserForm12.TextBox2.Text = Str(nom)
Load UserForm12
UserForm12.Show



End Sub
