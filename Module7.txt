' GEN_USL_x_DS — генерация условий связи штуцеров

Sub GEN_USL_x_DS()
    aa = Worksheets.Count
    chsv = 0

    Dim liss As Long, ii As Long, jj As Long
    Dim ws As Worksheet
    Dim data As Variant
    Dim outArr() As Variant
    Dim USL_x As String
    Dim aawV As Variant
    Dim k1 As Variant, k2 As Variant, k3 As Variant
    Dim kk1 As String, kk2 As String, kk3 As String
    Dim ind_k As Long
    Dim rBase As Long, cJ As Long

    ReDim outArr(1 To 19, 1 To 1)

    For liss = 3 To aa
        Set ws = Worksheets(liss)
        data = ws.Range(ws.Cells(5, 7), ws.Cells(133, 26)).Value  ' 129 x 20

        For ii = 1 To 19
            USL_x = "1"
            rBase = (ii - 1) * 7 + 1
            aawV = data(rBase, 1)
            If Len(aawV) > 1 Then
                For jj = 1 To 9
                    cJ = 4 + (jj - 1) * 2
                    k1 = data(rBase, cJ)
                    k2 = data(rBase + 1, cJ)
                    k3 = data(rBase + 2, cJ)
                    ind_k = Len(k1) + Len(k2) + Len(k3)
                    kk1 = "0"
                    If Len(k1) > 0 Then kk1 = k1
                    kk2 = ""
                    If Len(k2) > 0 Then kk2 = "+" + k2
                    kk3 = ""
                    If Len(k3) > 0 Then kk3 = "+" + k3
                    If ind_k > 0 Then
                        USL_x = USL_x + "*(" + kk1 + kk2 + kk3 + ")"
                        chsv = chsv + 1
                    End If
                Next jj
                If IsNumeric(USL_x) = True Then USL_x = Str(USL_x) + "*()"
            Else
                USL_x = ""
            End If
            outArr(ii, 1) = USL_x
        Next ii

        ' запись результата по адресам Cells(5+(ii-1)*7+4, 16)
        For ii = 1 To 19
            ws.Cells(5 + (ii - 1) * 7 + 4, 16).Value = outArr(ii, 1)
        Next ii

        Call PrPro_DS(100 * (liss - 2) / (aa - 2), "Библиотека. Условия связи штуцеров!")
    Next liss

    chsv = Format(100 * chsv / (Worksheets.Count - 2), 0#) / 100
    Worksheets("CODE").Cells(53, 12).Value = chsv
End Sub


' =====================================================================
' PrPro_DS — троттлинг UI (~30 fps)
' =====================================================================
Sub PrPro_DS(progr As Double, procName As String)
    Static lastT As Single
    If progr > 100 Then progr = 100
    If progr < 0 Then progr = 0
    If progr < 100 And (Timer - lastT) < 0.03 Then Exit Sub
    lastT = Timer
    ufProgress.Label3.Caption = ">>> " + procName
    ufProgress.Label2.Width = 0.01 + progr / 100 * ufProgress.FrameProgress.Width
    ufProgress.Label1.Caption = Str(Format(progr * 100, 0#) / 100) + "%"
    If InStr(ufProgress.Label1.Caption, ".") = 0 Then ufProgress.Label1.Caption = Replace(ufProgress.Label1.Caption, "%", ".0%")
    If InStr(ufProgress.Label1.Caption, ".") > 0 Then
        aa1 = Right(ufProgress.Label1.Caption, Len(ufProgress.Label1.Caption) - InStr(ufProgress.Label1.Caption, ".") + 1)
        If Len(aa1) < 4 Then
            ufProgress.Label1.Caption = Replace(ufProgress.Label1.Caption, "%", "0%")
        End If
    End If
    DoEvents
End Sub

    Dim src54 As Variant, src55 As Variant, src56 As Variant
    Dim src60_65 As Variant, src329_352 As Variant

    src54 = srcWs.Range(srcWs.Cells(63, 54), srcWs.Cells(iEnd, 54)).Value
    src55 = srcWs.Range(srcWs.Cells(63, 55), srcWs.Cells(iEnd, 55)).Value
    src56 = srcWs.Range(srcWs.Cells(63, 56), srcWs.Cells(iEnd, 56)).Value
    src60_65 = srcWs.Range(srcWs.Cells(63, 60), srcWs.Cells(iEnd, 65)).Value
    src329_352 = srcWs.Range(srcWs.Cells(63, 329), srcWs.Cells(iEnd, 352)).Value

    Dim out59_65() As Single
    Dim out329_352() As Single
    ReDim out59_65(1 To nRows, 1 To 7)
    ReDim out329_352(1 To nRows, 1 To 24)

    Dim i As Long, jj As Long
    Dim sumu As Single, sumu2 As Single
    Dim valArr(1 To 6) As Single

    For i = 1 To nRows
        sumu = 0
        sumu2 = 0

        For jj = 1 To 6
            valArr(jj) = Abs(src60_65(i, jj))
            sumu = valArr(jj) + sumu
        Next jj
        If sumu = 0 Then sumu = 0.0000000000001

        ' out59 всегда 0 (ui(0) = 0 в оригинале)
        out59_65(i, 1) = 0
        For jj = 1 To 6
            out59_65(i, jj + 1) = valArr(jj) / sumu
        Next jj

        For jj = 1 To 24
            sumu2 = Abs(src329_352(i, jj)) + sumu2
        Next jj
        If sumu2 = 0 Then sumu2 = 0.0000000000001

        For jj = 1 To 24
            out329_352(i, jj) = Abs(src329_352(i, jj)) / sumu2
        Next jj
    Next i

    dstWs.Range(dstWs.Cells(63, 54), dstWs.Cells(iEnd, 54)).Value = src54
    dstWs.Range(dstWs.Cells(63, 55), dstWs.Cells(iEnd, 55)).Value = src55
    dstWs.Range(dstWs.Cells(63, 56), dstWs.Cells(iEnd, 56)).Value = src56
    dstWs.Range(dstWs.Cells(63, 328), dstWs.Cells(iEnd, 328)).Value = src54
    dstWs.Range(dstWs.Cells(63, 59), dstWs.Cells(iEnd, 65)).Value = out59_65
    dstWs.Range(dstWs.Cells(63, 329), dstWs.Cells(iEnd, 352)).Value = out329_352

    For i = 63 To iEnd
        dstWs.Cells(i, 54).Interior.Color = srcWs.Cells(i, 54).Interior.Color
    Next i
End Sub


' =====================================================================
' PrPro_s3 — троттлинг UI (~30 fps). Совместим с исходным PrPro
' =====================================================================
Sub PrPro_s3(progr As Double, procName As String)
    Static lastT As Single
    If progr > 100 Then progr = 100
    If progr < 0 Then progr = 0
    If progr < 100 And (Timer - lastT) < 0.03 Then Exit Sub
    lastT = Timer
    ufProgress.Label3.Caption = ">>> " + procName
    ufProgress.Label2.Width = 0.01 + progr / 100 * ufProgress.FrameProgress.Width
    ufProgress.Label1.Caption = Str(Format(progr * 100, 0#) / 100) + "%"
    If InStr(ufProgress.Label1.Caption, ".") = 0 Then ufProgress.Label1.Caption = Replace(ufProgress.Label1.Caption, "%", ".0%")
    If InStr(ufProgress.Label1.Caption, ".") > 0 Then
        aa1 = Right(ufProgress.Label1.Caption, Len(ufProgress.Label1.Caption) - InStr(ufProgress.Label1.Caption, ".") + 1)
        If Len(aa1) < 4 Then
            ufProgress.Label1.Caption = Replace(ufProgress.Label1.Caption, "%", "0%")
        End If
    End If
    DoEvents
End Sub
