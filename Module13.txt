Public Function AhNorm(ByVal zz As String) As String
    ' NEWGEN: регистр/переносы/табы -> пробел, схлопывание кратных пробелов, трим.
    Dim kk As Long
    zz = UCase$(zz)
    zz = Replace(zz, Chr(13), " ")
    zz = Replace(zz, Chr(10), " ")
    zz = Replace(zz, vbTab, " ")
    zz = Replace(zz, Chr(160), " ") ' неразрывный пробел
    Do While InStr(zz, "  ") > 0
        zz = Replace(zz, "  ", " ")
    Loop
    AhNorm = Trim$(zz)
End Function

Public Function AhTok(ByVal zz As String) As String()
    ' NEWGEN: токены - последовательности букв/цифр/_; всё прочее - разделитель.
    Dim LL As Long, aa As String, out() As String, nn As Long
    zz = AhNorm(zz)
    nn = 0
    ReDim out(1 To 200)
    aa = ""
    For LL = 1 To Len(zz)
        cc = Mid$(zz, LL, 1)
        ' v2: добавлена кириллица (дефект v1: AhTok отбрасывал кириллицу - кириллические ключи не матчится)
        If (cc >= "A" And cc <= "Z") Or (cc >= ChrW(1040) And cc <= ChrW(1071)) Or cc = ChrW(1025) Or cc = ChrW(1105) Or (cc >= "0" And cc <= "9") Or cc = "_" Then
            aa = aa & cc
        Else
            If Len(aa) > 0 Then
                nn = nn + 1
                out(nn) = aa
                aa = ""
            End If
        End If
    Next LL
    If Len(aa) > 0 Then
        nn = nn + 1
        out(nn) = aa
    End If
    ReDim Preserve out(1 To nn + 1)
    out(nn + 1) = "" ' терминатор
    AhTok = out
End Function

Public Function AhLev(ByVal aa As String, ByVal bb As String) As Long
    ' NEWGEN: расстояние Левенштейна (классика, динамическое программирование).
    Dim nn As Long, mm As Long, ii As Long, jj As Long
    Dim dd() As Long, cost As Long
    nn = Len(aa): mm = Len(bb)
    If nn = 0 Then AhLev = mm: Exit Function
    If mm = 0 Then AhLev = nn: Exit Function
    ReDim dd(0 To nn, 0 To mm)
    For ii = 0 To nn: dd(ii, 0) = ii: Next ii
    For jj = 0 To mm: dd(0, jj) = jj: Next jj
    For ii = 1 To nn
        For jj = 1 To mm
            If Mid$(aa, ii, 1) = Mid$(bb, jj, 1) Then cost = 0 Else cost = 1
            If dd(ii - 1, jj) + 1 < dd(ii, jj - 1) + 1 Then
                dd(ii, jj) = dd(ii - 1, jj) + 1
            Else
                dd(ii, jj) = dd(ii, jj - 1) + 1
            End If
            If dd(ii - 1, jj - 1) + cost < dd(ii, jj) Then dd(ii, jj) = dd(ii - 1, jj - 1) + cost
        Next jj
    Next ii
    AhLev = dd(nn, mm)
End Function

Public Function AhEq(ByVal tok As String, ByVal kw As String) As Boolean
    ' NEWGEN: нечувствительное сравнение: точное | Лев.<=1 (len>=6) | Лев.<=2 (len>=10) |
    '         срезанный конец сообщения: токен и ключ совпадают по первым 5 символам при разнице <=3;
    '         ключ как префикс токена (ALL_GS внутри ALL_GS_COUNT) или наоборот - при len>=5.
    Dim ll1 As Long, ll2 As Long, ee As Long
    ll1 = Len(tok): ll2 = Len(kw)
    If ll1 = 0 Or ll2 = 0 Then Exit Function
    If tok = kw Then AhEq = True: Exit Function
    If ll1 >= 10 And ll2 >= 10 Then
        If AhLev(tok, kw) <= 2 Then AhEq = True: Exit Function
    ElseIf ll1 >= 6 And ll2 >= 6 Then
        If AhLev(tok, kw) <= 1 Then AhEq = True: Exit Function
    End If
    If ll1 >= 5 And ll2 >= 5 Then
        If Left$(tok, 5) = Left$(kw, 5) Then
            ee = Abs(ll1 - ll2)
            If ee <= 3 Then AhEq = True: Exit Function ' срезанный конец
        End If
    End If
    If ll1 >= 5 And ll2 >= 5 Then
        If ll1 > ll2 Then
            If Left$(tok, ll2) = kw Then AhEq = True: Exit Function ' ключ - префикс токена (ALL_GS в ALL_GS_COUNT)
        ElseIf ll1 < ll2 Then
            If Left$(kw, ll1) = tok Then AhEq = True: Exit Function ' токен - префикс ключа (срезан и хвост ключа)
        End If
    End If
End Function

Public Function AhClassify(ByVal msg As String) As String
    ' Возвращает код карточки или "" — по токенно-нечувствительному сопоставлению с ключами.
    ' Скоринг: каждая фраза-ключ, все слова которой найдены в тексте (fuzzy), даёт +1.
    ' Победитель — максимум совпавших фраз; при равенстве — максимум суммарной длины совпавших фраз.
    ' Порог: хотя бы 1 совпавшая фраза.
    Dim kls As String, kk() As String, ii As Long, jj As Long
    Dim best As String, bestsc As Long, bestlen As Long
    Dim ntok As Long
    Dim tks() As String
    tks = AhTok(msg)
    ntok = 0
    Do While tks(ntok + 1) <> ""
        ntok = ntok + 1
    Loop
    kls = AhKeyList()
    kk = Split(kls, "|")
    best = "": bestsc = 0: bestlen = 0
    ii = 0
    Do While ii < UBound(kk)
        Dim code As String
        code = kk(ii)
        jj = ii + 1
        Dim found As Long, flen As Long
        found = 0: flen = 0
        Do While jj <= UBound(kk)
            Dim kw As String
            kw = kk(jj)
            If IsCode(kw) Then Exit Do ' следующий код — конец фраз этого кода
            If AhPhraseIn(toks2(tks, ntok), kw) Then
                found = found + 1
                flen = flen + Len(kw)
            End If
            jj = jj + 1
        Loop
        If found > bestsc Or (found = bestsc And found > 0 And flen > bestlen) Then
            bestsc = found
            bestlen = flen
            best = code
        End If
        ii = jj
    Loop
    If bestsc >= 1 Then AhClassify = best Else AhClassify = ""
End Function

Private Function IsCode(ByVal zz As String) As Boolean
    ' Код — строка вида XXX_YYY (латиница, цифры, подчёркивания, без пробелов, длина 3-30).
    Dim ii As Long, cc As String, ok As Boolean
    If Len(zz) < 3 Or Len(zz) > 30 Then Exit Function
    ok = True
    For ii = 1 To Len(zz)
        cc = Mid$(zz, ii, 1)
        If Not ((cc >= "A" And cc <= "Z") Or (cc >= "0" And cc <= "9") Or cc = "_") Then ok = False: Exit For
    Next ii
    IsCode = ok
End Function

Private Function toks2(tks() As String, ByVal nn As Long) As String()
    Dim out() As String, ii As Long
    ReDim out(1 To nn)
    For ii = 1 To nn
        out(ii) = tks(ii)
    Next ii
    toks2 = out
End Function

Private Function AhPhraseIn(tks() As String, ByVal phr As String) As Boolean
    ' Все слова фразы найдены среди токенов (fuzzy AhEq).
    Dim words() As String, ii As Long, jj As Long, ok As Boolean
    words = Split(AhNorm(phr), " ")
    For ii = 0 To UBound(words)
        If Len(words(ii)) > 0 Then
            ok = False
            For jj = 1 To UBound(tks)
                If AhEq(tks(jj), words(ii)) Then ok = True: Exit For
            Next jj
            If Not ok Then Exit Function
        End If
    Next ii
    AhPhraseIn = True
End Function

Private Function CountEmpty(words() As String) As Long
    Dim nn As Long, cc As Long
    cc = 0
    For nn = 1 To UBound(words) + 1
        If Len(words(nn - 1)) = 0 Then cc = cc + 1
    Next nn
    CountEmpty = cc
End Function

' ---------- извлечение параметров из текста ----------
Private ep_has As Boolean
Private ep_sheet As String
Private ep_link As String
Private ep_num As Long
Private ep_arm As String
Private ep_unk As String
Private ep_addr As String

Public Sub AhExtract(ByVal msg As String)
    ' NEWGEN: параметры регулярными выражениями (позднее связывание, VBScript.RegExp).
    ' Заполняет модульные переменные ep_* (без UDT — он вызывал User-defined type not defined).
    Dim RX As Object, mch As Object, mm As Object
    ep_has = False
    ep_sheet = ""
    ep_link = ""
    ep_num = 0
    ep_arm = ""
    ep_unk = ""
    ep_addr = ""
    On Error Resume Next
    Set RX = CreateObject("VBScript.RegExp")
    If RX Is Nothing Then Exit Sub
    RX.Global = True
    RX.IgnoreCase = True
    ' имя листа: "на листе X" | "лист X"
    RX.Pattern = "(?:НА ЛИСТЕ|ЛИСТ)\s+([A-Za-z0-9_\-]+)"
    Set mch = RX.Execute(msg)
    If mch.Count > 0 Then ep_sheet = mch(0).SubMatches(0)
    ' имя связи: "связь X" или "в связи X"
    RX.Pattern = "СВЯЗЬ\s+([A-Za-z0-9_\-]+)"
    Set mch = RX.Execute(msg)
    If mch.Count > 0 Then ep_link = mch(0).SubMatches(0)
    ' номер связи: "связь №N"
    RX.Pattern = "СВЯЗЬ\s*№\s*([0-9]+)"
    Set mch = RX.Execute(msg)
    If mch.Count > 0 Then ep_num = CLng(Val(mch(0).SubMatches(0)))
    ' имена в <>: первое — арматура, второе — неизвестная часть
    RX.Pattern = "<([^<>]{2,80})>"
    Set mch = RX.Execute(msg)
    If mch.Count > 0 Then ep_arm = mch(0).SubMatches(0)
    If mch.Count > 1 Then ep_unk = mch(1).SubMatches(0)
    ' ячейки в формате (R+3,J) и (D7)
    Dim ss As String, qq As String
    ss = ""
    RX.Pattern = "\(R\+([0-9]),\s*([A-Z])\)"
    Set mch = RX.Execute(msg)
    For Each mm In mch
        If ep_num > 0 Then
            qq = AhCellAddr(ep_num, CLng(Val(mm.SubMatches(0))), mm.SubMatches(1))
        Else
            qq = "строка контактов + " & mm.SubMatches(0) & ", колонка " & mm.SubMatches(1)
        End If
        If Len(ss) > 0 Then ss = ss & "; "
        ss = ss & qq
    Next mm
    RX.Pattern = "\(([A-Z])([0-9]{1,3})\)"
    Set mch = RX.Execute(msg)
    For Each mm In mch
        If Len(ss) > 0 Then ss = ss & "; "
        ss = ss & mm.SubMatches(0) & mm.SubMatches(1)
    Next mm
    ep_addr = ss
    If Len(ep_sheet) > 0 Or Len(ep_link) > 0 Or Len(ep_arm) > 0 Or ep_num > 0 Then ep_has = True
End Sub

Public Function AhCellAddr(ByVal nlink As Long, ByVal offs As Long, ByVal colLetter As String) As String
    ' строка связи №nlink: 5 + 7*(nlink-1); +offs; колонка - буква.
    Dim rr As Long
    rr = 5 + 7 * (nlink - 1) + offs
    AhCellAddr = colLetter & CStr(rr)
End Function

' ---------- разбор текста на отдельные ошибки ----------
Public Function AhSplitMsgs(ByVal big As String) As String()
    ' Разбивает свод/файл на отдельные ошибки по маркерам «Ошибка №» и «Замечание №».
    ' Нормализация переносов, регистронезависимость, защита от срезанных/дубль-маркеров.
    Dim ss As String, out() As String, nn As Long, pp As Long, qq As Long
    ss = AhNorm(big)
    ReDim out(1 To 500)
    nn = 0
    pp = 1
    Do While pp <= Len(ss)
        qq = AiNextMarker(ss, pp)
        If qq = 0 Then Exit Do
        ' конец маркера - следующий маркер или конец
        Dim ee As Long
        ee = AiNextMarker(ss, qq + 12)
        If ee = 0 Then ee = Len(ss) + 1
        Dim txt As String
        txt = Mid$(ss, qq, ee - qq) ' NEWGEN: текст ошибки начинается с маркера (без шапки свода)
        If Len(Trim$(txt)) > 8 Then
            nn = nn + 1
            out(nn) = Trim$(txt)
        End If
        pp = ee
    Loop
    ReDim Preserve out(1 To nn + 1)
    out(nn + 1) = ""
    AhSplitMsgs = out
End Function

Private Function AiNextMarker(ByVal ss As String, ByVal fromPos As Long) As Long
    ' Ближайший из "ОШ№"/"ЗАМ№" начиная с fromPos (по AhNorm: "ОШИБКА №"/"ЗАМЕЧАНИЕ №").
    Dim a1 As Long, a2 As Long
    a1 = InStr(fromPos, ss, "ОШИБКА №")
    a2 = InStr(fromPos, ss, "ЗАМЕЧАНИЕ №")
    If a1 = 0 Then AiNextMarker = a2: Exit Function
    If a2 = 0 Then AiNextMarker = a1: Exit Function
    If a1 < a2 Then AiNextMarker = a1 Else AiNextMarker = a2
End Function

Public Function AhSourceText() As String

    ss = Trim$(Worksheets("CODE").Cells(6, 4).Value)

        AhSourceText = ss
        Exit Function
    End If

    Dim ff As String, nn As Long, best As String, bestt As String
    ff = Dir(ThisWorkbook.path & Application.PathSeparator & "_CONNECT_ERR_*.txt")
    best = ""
    Do While ff <> ""
        If ThisWorkbook.FileFormat <> 0 Then
            If ff > best Then best = ff ' имена содержат ММ_ДД - лексикографический порядок ок
        End If
        ff = Dir
    Loop
    If Len(best) > 0 Then
        Dim ts As Object, fso As Object
        Set fso = CreateObject("Scripting.FileSystemObject")

        AhSourceText = ts.ReadAll
        ts.Close
        Exit Function
    End If
    AhSourceText = ""
End Function

    End Select

    ah_s = ah_s & AhCtx(code)
    ah_s = ah_s & AhProbe(code)

    AhCard = ah_s
End Function

Public Function AhCtx(ByVal code As String) As String
    ' NEWGEN v2: источник замечания - из какого теста и какие места книги проверяются.
    Dim zz As String, kk() As String, ii As Long, out As String
    zz = AhCtxList()
    kk = Split(zz, "|")
    ii = 0
    Do While ii <= UBound(kk) - 3
        If StrComp(kk(ii), code, vbTextCompare) = 0 Then
            out = "ИСТОЧНИК ЗАМЕЧАНИЯ: тест " & kk(ii + 1) & " (строки кода " & kk(ii + 2) & "); проверяемые места: " & kk(ii + 3) & Chr(13) & Chr(10)
            Exit Do
        End If
        ii = ii + 4
    Loop
    If Len(out) = 0 Then out = "ИСТОЧНИК ЗАМЕЧАНИЯ: привязка к тесту не задана в v2 (смотрите карточку и чек-лист)." & Chr(13) & Chr(10)
    AhCtx = out
End Function

Private Function AhColNum(ByVal s As String) As Long
    ' A..ZZ -> номер колонки (для AhProbe).
    Dim ii As Long, vv As Long, ch As String
    vv = 0
    For ii = 1 To Len(s)
        ch = UCase$(Mid$(s, ii, 1))
        If ch >= "A" And ch <= "Z" Then vv = vv * 26 + (Asc(ch) - Asc("A") + 1)
    Next ii
    AhColNum = vv
End Function

Public Function AhProbe(ByVal code As String) As String
    ' NEWGEN v2: СПЕЦ-СБОР И АНАЛИЗ КНИГИ. По контексту (ep_*) и карте AhCtx собирает
    ' факты: есть ли лист, значение проверяемых ячеек, строка связи, арматура.
    ' Любая ошибка сбора - тихий пропуск (On Error Resume Next), разбор не ломается.
    Dim out As String, ii As Long, jj As Long, rr As Long, cc As Long
    Dim sh As Object, ws As Object, fnd As Object, addr As String
    out = ""
    On Error Resume Next
    If Len(ep_sheet) = 0 And Len(ep_addr) = 0 And ep_num = 0 And Len(ep_arm) = 0 Then
        AhProbe = out
        Exit Function
    End If
    Set sh = Nothing
    If Len(ep_sheet) > 0 Then
        For ii = 1 To ThisWorkbook.Worksheets.Count
            If StrComp(ThisWorkbook.Worksheets(ii).Name, ep_sheet, vbTextCompare) = 0 Then
                Set sh = ThisWorkbook.Worksheets(ii)
                Exit For
            End If
        Next ii
    End If
    If Len(ep_sheet) > 0 Then
        out = out & "СБОР ИЗ КНИГИ:" & Chr(13) & Chr(10)
        If sh Is Nothing Then
            out = out & " - лист <" & ep_sheet & "> НЕ НАЙДЕН в книге - проверьте имя листа (регистр/пробелы/типографика)." & Chr(13) & Chr(10)
        Else
            out = out & " - лист <" & ep_sheet & "> есть, строк: " & sh.UsedRange.rows.Count & "." & Chr(13) & Chr(10)
        End If
    End If
    ' --- значения адресов из текста ошибки и карты контекста ---
    Set fnd = CreateObject("VBScript.RegExp")
    fnd.Global = True
    fnd.Pattern = "[A-Z]{1,2}[0-9]{1,4}"
    Dim src As String, adrs As Collection
    Set adrs = New Collection
    If Len(ep_addr) > 0 Then adrs.Add ep_addr
    If Not sh Is Nothing Then
        If Len(tc_out) > 0 Then
            Dim mm As Object, ee As Object
            For Each ee In fnd.Execute(tc_out)
                If adrs.Count < 8 Then adrs.Add ee.Value
            Next ee
        End If
    End If
    If Not sh Is Nothing Then
        For ii = 1 To adrs.Count
            addr = adrs(ii)
            rr = 0: cc = 0
            For jj = 1 To Len(addr)
                If Mid$(addr, jj, 1) >= "0" And Mid$(addr, jj, 1) <= "9" Then
                    rr = CInt(Mid$(addr, jj))
                    cc = AhColNum(Left$(addr, jj - 1))
                    Exit For
                End If
            Next jj
            If rr > 0 And cc > 0 Then
                val2 = sh.Cells(rr, cc).Value
                If Len(Trim$(CStr(val2))) = 0 Then
                    out = out & " - ячейка " & addr & " = <пусто> (вероятная причина: незаполнено)." & Chr(13) & Chr(10)
                ElseIf IsNumeric(val2) Then
                    out = out & " - ячейка " & addr & " = " & Format$(val2, "0.####") & " (число)." & Chr(13) & Chr(10)
                Else
                    out = out & " - ячейка " & addr & " = <" & CStr(val2) & "> - ТЕКСТ вместо числа (вероятная причина ошибки формата)." & Chr(13) & Chr(10)
                End If
            End If
        Next ii
    End If
    ' --- строка связи ---
    If ep_num > 0 And Not sh Is Nothing Then
        rr = 5 + 7 * (ep_num - 1)
        Dim sL As String
        sL = ""
        For cc = 1 To 10
            If Len(Trim$(CStr(sh.Cells(rr, cc).Value))) > 0 Then sL = sL & Chr(64 + cc) & rr & "=" & CStr(sh.Cells(rr, cc).Value) & "; "
        Next cc
        out = out & " - связь " & ep_num & ": строка " & rr & " листа <" & ep_sheet & ">; данные: " & sL & Chr(13) & Chr(10)
    End If
    ' --- арматура: первое вхождение по книге ---
    If Len(ep_arm) > 0 Then
        Dim hit As Boolean, hitSh As String, hitAd As String, hitVal As String
        hit = False
        For ii = 1 To ThisWorkbook.Worksheets.Count
            Set fnd = ThisWorkbook.Worksheets(ii).UsedRange.Find(ep_arm, LookIn:=xlValues, LookAt:=xlPart)
            If Not fnd Is Nothing Then
                hit = True
                hitSh = ThisWorkbook.Worksheets(ii).Name
                hitAd = fnd.Address(False, False)
                hitVal = CStr(fnd.Offset(0, -1).Value)
                Exit For
            End If
        Next ii
        If hit Then
            out = out & " - арматура <" & ep_arm & "> найдена: лист <" & hitSh & "> ячейка " & hitAd & "; слева: " & hitVal & "." & Chr(13) & Chr(10)
        Else
            out = out & " - арматура <" & ep_arm & "> НЕ НАЙДЕНА ни на одном листе - вероятно опечатка или арматура не заведена." & Chr(13) & Chr(10)
        End If
    End If
    AhProbe = out
End Function
' ---------- главный вход ----------
Public Sub AI_Help()
    ' Вход ассистента: InputBox с текстом ошибки; пусто - свод CODE(6,4) или последний _CONNECT_ERR_*.
    Dim big As String, ss As String
    ss = InputBox("Вставь текст ошибки/замечания (можно несколько подряд)." & Chr(13) + Chr(10) & _
        "Пусто - возьму свод CODE!(6,4) или последний файл _CONNECT_ERR рядом с книгой.", "ИИ-ассистент разборов ошибок", "")
    If Len(Trim$(ss)) > 0 Then
        big = ss
    Else
        big = AhSourceText()
        If Len(big) = 0 Then
            MsgBox "Не введён текст и не найден свод ошибок (CODE!(6,4) пуст, файлов _CONNECT_ERR нет).", vbInformation, "ИИ-ассистент"
            Exit Sub
        End If
    End If
    AI_Process big
End Sub

Public Sub AI_Process(ByVal big As String)
    ' Разбор свода, классификация, вывод в CODE!(6,32).
    ' NEWGEN: листы НЕ создаются (число листов в книге жёстко привязано к источникам) - отчёт пишется
    ' в одну ячейку CODE!(6,32): полный текст с переносами строк, WrapText включён.
    Dim msgs() As String, nn As Long, kk As Long
    Dim code As String, card As String
    Dim rep As String
    msgs = AhSplitMsgs(big)
    nn = 0
    Do While msgs(nn + 1) <> ""
        nn = nn + 1
    Loop
    If nn = 0 Then
        MsgBox "Не найдено ни одной ошибки в тексте.", vbInformation, "ИИ-ассистент"
        Exit Sub
    End If
    rep = "ИИ-АССИСТЕНТ РАЗБОРОВ ОШИБОК - отчёт от " & Format(Now, "dd.mm.yyyy hh:nn:ss")
    kk = 0
    Do While kk < nn
        kk = kk + 1
        Call AhExtract(msgs(kk))
        code = AhClassify(msgs(kk))
        card = AhCard(code)
        rep = rep & Chr(13) + Chr(10) & "---- ОШИБКА " & kk & " ----"
        rep = rep & Chr(13) + Chr(10) & Replace(card, Chr(13) + Chr(10), vbLf)
        rep = rep & Chr(13) + Chr(10) & "ТЕКСТ ОРИГИНАЛА: " & msgs(kk)
        rep = rep & Chr(13) + Chr(10)
    Loop
    With Worksheets("CODE")
        .Cells(6, 32).Value = rep
        .Cells(6, 32).WrapText = True
        .Cells(6, 32).VerticalAlignment = xlTop
    End With
    MsgBox "Обработано ошибок: " & nn & ". Отчёт - в CODE! ячейка 6,32 (AF6).", vbInformation, "ИИ-ассистент"
End Sub

' ---------- копирование всего отчёта в буфер ----------
Public Sub AI_CopyAll()
    ' NEWGEN: копирует отчёт из CODE!(6,32) в буфер обмена.
    Dim ss As String
    ss = Worksheets("CODE").Cells(6, 32).Value
    If Len(ss) = 0 Then
        MsgBox "Отчёт пуст - сначала запусти AI_Help.", vbInformation, "ИИ-ассистент"
        Exit Sub
    End If
    Dim dob As Object
    Set dob = CreateObject("New:{1C3B4210-F441-11CE-B9EA-00AA006D1CC3}")
    dob.SetText ss
    dob.PutInClipboard
    MsgBox "Отчёт скопирован в буфер обмена.", vbInformation, "ИИ-ассистент"
End Sub
