Sub ipres() ' заполняет персетами каждый источник
gYes = 0
 answer = MsgBox("Заполнение пресетов для всех точек из единых КЛИШЕ!!!" + Chr(13) + Chr(10) + "Удалить все ранее установленные пресеты и заполнить из КЛИШЕ?", vbQuestion + vbYesNo + vbDefaultButton2, "Внимание!")
If answer = vbYes Then gYes = 1
If gYes = 1 Then
If Worksheets("CODE").Cells(5, 58).Value < 2 Then Application.ScreenUpdating = False
Worksheets("CODE").Activate
Range("BC63:BD2500").ClearContents
Range("BF63:CB2500").ClearContents
' считаем число источников
i = 0
b = 0
II0 = 0
Do
a = Cells(63 + i, 54).Value
If a = "" Then
b = 1
Else
i = i + 1
If Worksheets("CODE").Cells(62 + i, 54).Interior.Color = RGB(200, 50, 50) Then
II0 = II0 + 1
End If
End If
Loop While b = 0
chi = i


'Cells(60, 49).Value = dd
For i = 1 To chi

Worksheets("CODE").Cells(62 + i, 55).Value = Format((Worksheets("CODE").Cells(3, 55).Value + Worksheets("CODE").Cells(3, 55).Value * (5 - Rnd(1) * 10) / 1000) * 1000000, 0#) / 1000000 ' давления
Worksheets("CODE").Cells(62 + i, 56).Value = Worksheets("CODE").Cells(5, 55).Value ' температура
Worksheets("CODE").Cells(62 + i, 59).Value = Worksheets("CODE").Cells(6, 55).Value ' уровень
For j = 1 To 21
Worksheets("CODE").Cells(62 + i, 59 + j).Value = Worksheets("CODE").Cells(6 + j, 55).Value ' свойства
If Worksheets("CODE").Cells(6 + j, 55).Value = "" Then Worksheets("CODE").Cells(62 + i, 59 + j).Value = ""
If Worksheets("CODE").Cells(6 + j, 51).Value = "" Then Worksheets("CODE").Cells(62 + i, 59 + j).Value = ""
Next j
Next i


' считаем число кранов
i = 0
b = 0
Do
a = Cells(63 + i, 57).Value
If a = "" Then
b = 1
Else
i = i + 1
End If
Loop While b = 0
chK = i


For i = 1 To chK
Worksheets("CODE").Cells(62 + i, 58).Value = 1 ' клапана
Next i


aa = Worksheets.Count
shsv = 0
For i = 1 To aa
bb = Worksheets(i).Name
If Not bb = "CODE" Then
If Not bb = "EX" Then
For j = 5 To 131 Step 7
If Len(Worksheets(bb).Cells(j, 7).Value) > 1 Then
shsv = shsv + 1
End If
Next j
End If
End If
Next i
shsv = shsv - (chi - II0)

For i = 87 To 109
If Len(Cells(i, 19).Value) > 2 Then shsv = shsv + 1
Next i

Cells(57, 56).Value = "*-число точек связи " + Str(shsv)
Cells(58, 56).Value = "*-число листов с источниками " + Str(Worksheets.Count - 2)
Cells(59, 56).Value = "*-число источников " + Str(chi) + " (" + Str(II0) + "-безусл., " + Str(chi - II0) + "-усл.)"
Cells(60, 56).Value = "*-число арматур " + Str(chK)

End If ' gYes
Application.ScreenUpdating = True
End Sub
